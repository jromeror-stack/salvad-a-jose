  <script>
    // --- mini util ---
    const $ = (id) => document.getElementById(id);
    const toast = $("toast");
    const toastTitle = $("toastTitle");
    const toastMsg = $("toastMsg");

    function showToast(title, msg){
      toastTitle.textContent = title;
      toastMsg.textContent = msg;
      toast.classList.add("show");
      clearTimeout(showToast._t);
      showToast._t = setTimeout(() => toast.classList.remove("show"), 2400);
    }

    // --- estado "vivo" ---
    let carga = 76;
    let exc = 2;
    let firmas = 128;

    function clamp(n,min,max){ return Math.max(min, Math.min(max,n)); }

    // simula que el tiempo pasa y la carga sube (porque siempre sube)
    setInterval(() => {
      carga = clamp(carga + (Math.random() < 0.6 ? 1 : 0), 55, 99);
      $("carga").textContent = carga;
      $("mins").textContent = String(clamp(parseInt($("mins").textContent,10) + 1, 1, 59));
    }, 6000);

    // --- excusas pro ---
    const excusas = [
      "Estoy en una call con stakeholders (el lavavajillas).",
      "Ahora mismo estoy bloqueado por una dependencia externa (la fregona no responde).",
      "Voy a hacer un quick sync de 10 minutos (en el sofá).",
      "Estoy cerrando un incidente crítico (mi salud mental).",
      "He puesto un change freeze: no se aceptan nuevas tareas hasta mañana.",
      "Estoy en modo análisis: si me muevo, rompo el experimento.",
      "Estoy compilando… mi paciencia."
    ];

    function nuevaExcusa(){
      const pick = excusas[Math.floor(Math.random()*excusas.length)];
      $("excusaBox").textContent = pick;
      showToast("Excusa generada", "Úsala con voz tranquila y mirada al infinito.");
    }

    // --- acciones ---
    $("btnRescate").addEventListener("click", () => {
      carga = clamp(carga - 8, 10, 99);
      exc = clamp(exc + 1, 0, 5);
      $("carga").textContent = carga;
      $("exc").textContent = exc;
      showToast("Protocolo activado", "Se ha asignado un escudo anti ‘ya que estás…’ (temporal).");
    });

    $("btnAplazar").addEventListener("click", () => {
      carga = clamp(carga + 3, 10, 99);
      $("carga").textContent = carga;
      showToast("Aplazado", "La culpa ha sido programada para las 21:37.");
    });

    $("btnModoSigilo").addEventListener("click", () => {
      showToast("Modo sigilo", "Auriculares puestos. Frase clave: “lo veo y te digo”.");
    });

    $("btnFirmar").addEventListener("click", () => {
      firmas += 1 + Math.floor(Math.random()*3);
      $("firmas").textContent = firmas;
      showToast("Firmado", "Se ha registrado tu apoyo a los 20 minutos legales de sofá.");
    });

    $("btnDonarTiempo").addEventListener("click", () => {
      carga = clamp(carga - 12, 10, 99);
      $("carga").textContent = carga;
      showToast("Donación aceptada", "Has donado 10 minutos de ayuda real. Eso sí cambia el mundo.");
    });

    $("btnCrearExcusa").addEventListener("click", nuevaExcusa);

    // inicial
    nuevaExcusa();
  </script>
